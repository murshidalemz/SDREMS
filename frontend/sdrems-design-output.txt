function RescueApplication({ category = "volunteer" }) {
  const { user } = useAuth();
  const navigate = useNavigate();
  const isProfessional = category === "professional";

  const [step, setStep] = useState(1);
  const [error, setError] = useState("");

  const [form, setForm] = useState({
    ...emptyForm,
    fullName: user?.name || "",
    email: user?.email || "",
  });

  const [documents, setDocuments] = useState({
    identityProof: null,
    trainingCertificate: null,
    professionalId: null,
  });

  const updateField = (e) => {
    const { name, value, checked, type } = e.target;

    setForm((previous) => ({
      ...previous,
      [name]: type === "checkbox" ? checked : value,
    }));
  };

  const toggleSkill = (skill) => {
    setForm((previous) => ({
      ...previous,
      skills: previous.skills.includes(skill)
        ? previous.skills.filter((item) => item !== skill)
        : [...previous.skills, skill],
    }));
  };

  const handleFile = (e) => {
    const { name, files } = e.target;
    const file = files?.[0];

    if (!file) return;

    const allowedTypes = [
      "image/png",
      "image/jpeg",
      "application/pdf",
    ];

    if (!allowedTypes.includes(file.type)) {
      setError("Only PNG, JPG, JPEG and PDF files are allowed.");
      e.target.value = "";
      return;
    }

    if (file.size > 5 * 1024 * 1024) {
      setError("Each document must be 5 MB or smaller.");
      e.target.value = "";
      return;
    }

    setError("");
    setDocuments((previous) => ({
      ...previous,
      [name]: file,
    }));
  };

  const removeFile = (name) => {
    setDocuments((previous) => ({
      ...previous,
      [name]: null,
    }));
  };

  const validateStep = () => {
    setError("");

    if (step === 1) {
      const requiredFields = [
        "fullName",
        "phone",
        "dateOfBirth",
        "address",
        "state",
        "district",
        "city",
        "pinCode",
        "operatingLocation",
        "serviceArea",
        "availability",
        "emergencyContact",
      ];

      const missing = requiredFields.some(
        (field) => !String(form[field]).trim()
      );

      if (missing) {
        setError("Please complete all required personal information fields.");
        return false;
      }

      if (!/^\d{6}$/.test(form.pinCode)) {
        setError("Enter a valid 6-digit PIN code.");
        return false;
      }

        setError("Enter a valid 10-digit mobile number.");
